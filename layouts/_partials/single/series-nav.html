{{- /* 시리즈 안에서의 이전 편 / 다음 편 링크 (본문 바로 아래) */ -}} {{- $page := . -}} {{- range .GetTerms "series" -}} {{- $pages := .Pages.ByDate -}} {{- $prev := false -}} {{- $next := false -}} {{- range $i, $p := $pages -}} {{- if eq $p $page -}} {{- if gt $i 0 -}}{{- $prev = index $pages (sub $i 1) -}}{{- end -}} {{- if lt (add $i 1) (len $pages) -}}{{- $next = index $pages (add $i 1) -}}{{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{- with $prev -}} ← 이전 편{{ .Title }} {{- end -}} {{- with $next -}} 다음 편 →{{ .Title }} {{- end -}}
{{- end -}} {{- end -}}
